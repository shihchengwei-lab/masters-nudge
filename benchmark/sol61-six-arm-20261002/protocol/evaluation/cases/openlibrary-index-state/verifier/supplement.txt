

def test_mn_update_state_defaults_are_independent():
    first = SolrUpdateState()
    second = SolrUpdateState(commit=True)
    first.keys.append('/works/OL101W')
    first.adds.append({'key': '/works/OL101W'})
    first.deletes.append('/works/OL102W')
    assert second.keys == []
    assert second.adds == []
    assert second.deletes == []
    assert first.commit is False
    assert second.commit is True


def test_mn_update_state_combines_without_mutating_inputs():
    first = SolrUpdateState(keys=['one'], adds=[{'key': 'one'}])
    second = SolrUpdateState(keys=['two'], deletes=['two'], commit=True)
    combined = first + second
    assert combined.keys == ['one', 'two']
    assert combined.adds == [{'key': 'one'}]
    assert combined.deletes == ['two']
    assert combined.commit is True
    assert first.keys == ['one'] and first.deletes == [] and first.commit is False
    assert second.keys == ['two'] and second.adds == []


def test_mn_update_state_clear_preserves_keys_and_commit():
    state = SolrUpdateState(keys=['one'], adds=[{'key': 'one'}], deletes=['two'], commit=True)
    assert state.has_changes() is True
    state.clear_requests()
    assert state.has_changes() is False
    assert state.keys == ['one'] and state.commit is True
    assert state.adds == [] and state.deletes == []
